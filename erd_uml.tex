\documentclass{article}
% Pass the official PlantUML web server URL to render remotely
\usepackage[server=http://www.plantuml.com/plantuml]{plantuml}
\usepackage{caption}
\usepackage[T1]{fontenc}
\usepackage{pdflscape} % Rotates the page view in PDF readers
\usepackage{geometry}  % Allows adjusting page margins

\begin{document}

\newgeometry{margin=1.5cm} % Tight 1.5cm margins for maximum printable space
\begin{landscape}
\begin{figure}[p]          % 'p' places the figure on its own dedicated page
\centering
\begin{plantuml}
@startuml
skinparam classAttributeIconSize 0
hide circle
show spots

' Define Entity Styling
entity "repository" as repository {
  + repository_id : INTEGER <<PK>>
  --
  + name : TEXT
  + url : TEXT
  + packages : TEXT
  + status : TEXT
}

entity "developer" as developer {
  + developer_id : INTEGER <<PK>>
  --
  + name : TEXT
  + email : TEXT
  + login : TEXT
}

entity "developer_commit" as developer_commit {
  + commit_id : TEXT <<PK>>
  --
  + authored_date : TEXT
  + number_of_repo_packages : INTEGER
  + number_of_repo_files : INTEGER
  + number_of_repo_classes : INTEGER
  + number_of_repo_methods : INTEGER
  + status : TEXT
  --
  + repository_id : INTEGER <<FK>>
  + developer_id : INTEGER <<FK>>
}

entity "commit_file" as commit_file {
  + commit_file_id : INTEGER <<PK>>
  --
  + name : TEXT
  + amendment_type : TEXT
  --
  + file_id : INTEGER <<FK>>
  + commit_id : TEXT <<FK>>
  + repository_id : INTEGER <<FK>>
}

entity "commit_class" as commit_class {
  + commit_class_id : INTEGER <<PK>>
  --
  + name : TEXT
  + amendment_type : TEXT
  --
  + class_id : INTEGER <<FK>>
  + file_id : INTEGER <<FK>>
  + commit_id : TEXT <<FK>>
  + repository_id : INTEGER <<FK>>
}

entity "commit_method" as commit_method {
  + commit_method_id : INTEGER <<PK>>
  --
  + name : TEXT
  + amendment_type : TEXT
  --
  + method_id : INTEGER <<FK>>
  + class_id : INTEGER <<FK>>
  + commit_id : TEXT <<FK>>
  + repository_id : INTEGER <<FK>>
}

' External Entities referenced by Foreign Keys
entity "file" as file {
  + file_id : INTEGER <<PK>>
  + repository_id : INTEGER <<FK>>
}

entity "class" as class {
  + class_id : INTEGER <<PK>>
  + file_id : INTEGER <<FK>>
  + repository_id : INTEGER <<FK>>
}

entity "method" as method {
  + method_id : INTEGER <<PK>>
  + class_id : INTEGER <<FK>>
  + repository_id : INTEGER <<FK>>
}

' Relationships & Cardinalities
repository ||--o{ developer_commit : "1 to 0..*"
developer  ||--o{ developer_commit : "1 to 0..*"

developer_commit ||--o{ commit_file   : "1 to 0..*"
repository       ||--o{ commit_file   : "1 to 0..*"
file             ||--o{ commit_file   : "1 to 0..*"

developer_commit ||--o{ commit_class  : "1 to 0..*"
repository       ||--o{ commit_class  : "1 to 0..*"
file             ||--o{ commit_class  : "1 to 0..*"
class            ||--o{ commit_class  : "1 to 0..*"

developer_commit ||--o{ commit_method : "1 to 0..*"
repository       ||--o{ commit_method : "1 to 0..*"
class            ||--o{ commit_method : "1 to 0..*"
method           ||--o{ commit_method : "1 to 0..*"

file             ||--o{ class  : "1 to 0..*"
class            ||--o{ method : "1 to 0..*"
@enduml
\end{plantuml}
\end{figure}
\end{landscape}
\restoregeometry % Restores standard document margins for subsequent pages

\end{document}
